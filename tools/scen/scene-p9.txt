# Phase 9 (v1.1) Sable Ship scenes (x 6000..7320). Flat world: ground top at y=-60. Needs GRADLE_ARGS=-PwithSable.
# Ships: hollow 7x5x7 stone boxes turned into Sable sub-levels with /sable assemble area, named with /sable name set @n <name> positioned at the box centre
# (RCON runs at world spawn; @n = nearest pose position to the source). p9_metrics S4 checks each named ship starts at its box centre.
# Floating ships (bottom y=-50, 10 above the floor: out of Copper Reach 4) are held in place by the dev command /tentaworks ship <name> drive 0 0 0.
gamerule doDaylightCycle false
gamerule doWeatherCycle false
gamerule doImmediateRespawn true
time set noon
forceload add 6000 -20 6110 20
forceload add 6090 -20 6360 40

# W world baseline: wall x=6020, 20 high (idle hang Air)
fill 6020 -60 -6 6020 -41 6 stone
# A parked ship on the floor, turned 30 degrees about +Y (rotation point = its centre ~6063.5 -57.5 0.5)
fill 6060 -60 -3 6066 -56 3 stone hollow
sable assemble area 6060 -60 -3 6066 -56 3
execute positioned 6063.5 -57.5 0.5 run sable name set @n p9a
